
name: CI/CD - Api Gateway

on:
  push:
    branches:
      - main
    paths:
      - 'code/api-gateway/**'
      # - '.github/workflows/**'
  workflow_dispatch:

permissions:
  contents: read
  security-events: write

jobs:

  #* =================================*
  #* Stage: Gitleaks Secret Scan
  #* =================================*
  gitleaks-scan:
    name: Gitleaks Scan
    runs-on: ubuntu-latest

    steps:
      - name: Checkout source code
        uses: actions/checkout@v4
        with:
          fetch-depth: 0

      - name: Run Gitleaks Scan
        run: |
          docker run --rm \
            -v "${{ github.workspace }}:/repo" \
            zricethezav/gitleaks:latest detect \
            --source=/repo \
            --report-format=json \
            --report-path=/repo/gitleaks-report.json \
            --redact \
            --exit-code 0

      - name: Upload Gitleaks report
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: api-gateway-gitleaks-report
          path: gitleaks-report.json
          if-no-files-found: warn

  #* =================================*
  #* Stage: Checkov Dockerfile Scan
  #* =================================*
  checkov-dockerfile:
    name: Checkov Dockerfile Scan
    runs-on: ubuntu-latest
    needs: gitleaks-scan

    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-python@v5
        with:
          python-version: '3.11'

      - name: Install Checkov
        run: pip install checkov

      - name: Run Checkov Dockerfile scan
        run: |
          checkov -d code/api-gateway \
            --framework dockerfile \
            --output json > checkov-dockerfile-report.json || true

      - name: Upload report
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: api-gateway-checkov-dockerfile-report
          path: checkov-dockerfile-report.json
          if-no-files-found: warn

  #* =================================*
  #* Stage: Trivy Filesystem Scan
  #* =================================*
  trivy-scan:
    name: Trivy FS Scan - api-gateway
    runs-on: ubuntu-latest
    needs: gitleaks-scan

    steps:
      - uses: actions/checkout@v4

      - name: Verify Trivy container
        run: docker run --rm aquasec/trivy:0.68.2 --version

      - name: Run Trivy FS scan (JSON)
        run: |
          docker run --rm \
            -v "${{ github.workspace }}:/work" \
            aquasec/trivy:0.68.2 fs \
            --format json \
            --output /work/api-gateway-trivy-fs-report.json \
            --scanners vuln,secret,misconfig \
            --severity HIGH,CRITICAL \
            --ignore-unfixed \
            --exit-code 0 \
            /work/code/api-gateway

      - name: Run Trivy FS scan (TABLE)
        run: |
          docker run --rm \
            -v "${{ github.workspace }}:/work" \
            aquasec/trivy:0.68.2 fs \
            --format table \
            --scanners vuln,secret,misconfig \
            --severity HIGH,CRITICAL \
            --ignore-unfixed \
            --exit-code 0 \
            /work/code/api-gateway \
            > api-gateway-trivy-fs-report.txt

      - name: Upload Trivy JSON report
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: api-gateway-trivy-fs-json-report
          path: api-gateway-trivy-fs-report.json
          if-no-files-found: warn

      - name: Upload Trivy table report
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: api-gateway-trivy-fs-table-report
          path: api-gateway-trivy-fs-report.txt
          if-no-files-found: warn

  #* =================================*
  #* Stage: API Gateway Lint
  #* =================================*
  lint:
    name: Lint - api-gateway
    runs-on: ubuntu-latest
    needs:
      - trivy-scan
      - checkov-dockerfile

    defaults:
      run:
        working-directory: code/api-gateway

    steps:
      - uses: actions/checkout@v4

      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: '22'
          cache: npm
          cache-dependency-path: code/api-gateway/package-lock.json

      - name: Install dependencies
        run: npm ci

      - name: Run lint
        run: npm run lint --if-present
        continue-on-error: true

  #* =================================*
  #* Stage: API Gateway Tests
  #* =================================*
  api-gateway-test:
    name: Server Test
    runs-on: ubuntu-latest
    needs: lint

    defaults:
      run:
        working-directory: code/api-gateway

    steps:
      - name: Checkout source code
        uses: actions/checkout@v4

      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: '22'
          cache: npm
          cache-dependency-path: code/api-gateway/package-lock.json

      - name: Install dependencies
        run: npm ci

      - name: Run tests
        run: |
          if node -e "process.exit(require('./package.json').scripts?.test ? 0 : 1)"; then
            if grep -q 'no test specified' package.json; then
              echo "No tests configured yet. Skipping tests."
            else
              npm test
            fi
          else
            echo "No test script found. Skipping tests."
          fi

  #* =================================*
  #* Stage: SonarQube Analysis
  #* =================================*
  sonarqube:
    name: SonarQube Analysis - api-gateway
    runs-on: ubuntu-latest
    needs: api-gateway-test

    steps:
      - uses: actions/checkout@v4
        with:
          fetch-depth: 0

      - name: SonarQube Scan
        uses: sonarsource/sonarqube-scan-action@master
        with:
          projectBaseDir: code/api-gateway
        env:
          SONAR_TOKEN: ${{ secrets.SONAR_TOKEN }}
          SONAR_HOST_URL: ${{ secrets.SONAR_HOST_URL }}

      - name: SonarQube Quality Gate Check
        id: quality-gate
        continue-on-error: true
        uses: sonarsource/sonarqube-quality-gate-action@master
        with:
          scanMetadataReportFile: code/api-gateway/.scannerwork/report-task.txt
          pollingTimeoutSec: 600
        env:
          SONAR_TOKEN: ${{ secrets.SONAR_TOKEN }}
          SONAR_HOST_URL: ${{ secrets.SONAR_HOST_URL }}

      - name: Display Quality Gate Status
        if: always()
        run: |
          echo "Service: api-gateway"
          echo "Quality Gate status: ${{ steps.quality-gate.outputs.quality-gate-status }}"

  #* =================================*
  #* Stage: Docker Build, Scan and Push
  #* =================================*
  docker:
    name: API Gateway Docker Build, Scan & Push
    runs-on: ubuntu-latest
    needs: sonarqube

    steps:
      - name: Checkout source code
        uses: actions/checkout@v4

      #* Login to Docker Hub
      - name: Login to Docker Hub
        uses: docker/login-action@v4
        with:
          username: ${{ vars.DOCKERHUB_USERNAME }}
          password: ${{ secrets.DOCKERHUB_TOKEN }}

      #* Configure Docker Buildx
      - name: Set up QEMU
        uses: docker/setup-qemu-action@v4

      - name: Set up Docker Buildx
        uses: docker/setup-buildx-action@v4

      #* =================================*
      #* Stage: Build API Gateway Image
      #* =================================*
      - name: Build api-gateway Docker image
        uses: docker/build-push-action@v7
        with:
          context: ./code/api-gateway
          file: ./code/api-gateway/Dockerfile
          load: true
          push: false
          tags: sanketmahajan/micro-service-mernapp:api-gateway

      - name: Verify Docker image
        run: |
          docker image inspect sanketmahajan/micro-service-mernapp:api-gateway

      #* =================================*
      #* Stage: Trivy Image Security Scan
      #* =================================*
      - name: Save Docker image for Trivy
        run: |
          docker save \
            sanketmahajan/micro-service-mernapp:api-gateway \
            -o api-gateway-image.tar

      - name: Run Trivy image scan (JSON)
        run: |
          docker run --rm \
            -v "${{ github.workspace }}:/scan" \
            aquasec/trivy:0.68.2 image \
            --input /scan/api-gateway-image.tar \
            --format json \
            --output /scan/api-gateway-image-trivy-report.json \
            --severity HIGH,CRITICAL \
            --ignore-unfixed \
            --exit-code 0

      - name: Run Trivy image scan (TABLE)
        run: |
          docker run --rm \
            -v "${{ github.workspace }}:/scan" \
            aquasec/trivy:0.68.2 image \
            --input /scan/api-gateway-image.tar \
            --format table \
            --severity HIGH,CRITICAL \
            --ignore-unfixed \
            --exit-code 0 \
            > api-gateway-image-trivy-report.txt

      - name: Upload Trivy image JSON report
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: api-gateway-image-trivy-json-report
          path: api-gateway-image-trivy-report.json
          if-no-files-found: warn

      - name: Upload Trivy image TABLE report
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: api-gateway-image-trivy-table-report
          path: api-gateway-image-trivy-report.txt
          if-no-files-found: warn

      #* =================================*
      #* Stage: Generate Software Bill of Materials
      #* =================================*
      - name: Generate api-gateway source SBOM
        uses: anchore/sbom-action@v0
        with:
          path: ./code/api-gateway
          format: spdx-json
          output-file: api-gateway-source-sbom.spdx.json
          upload-artifact: true

      - name: Generate api-gateway image SBOM
        uses: anchore/sbom-action@v0
        with:
          image: sanketmahajan/micro-service-mernapp:api-gateway
          format: spdx-json
          output-file: api-gateway-image-sbom.spdx.json
          upload-artifact: true

      #* =================================*
      #* Stage: Push Docker Image
      #* =================================*
      - name: Push api-gateway Docker image
        run: |
          docker push sanketmahajan/micro-service-mernapp:api-gateway
